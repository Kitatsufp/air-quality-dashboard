function renderChart(metric, sPoints, pPoints, rPoints) { const isMain = metric === 'iaq'; const overlay = document.getElementById(`${metric}Overlay`); const noData = sPoints.length === 0 && pPoints.length === 0 && rPoints.length === 0; overlay.style.display = noData ? 'flex' : 'none'; const ctx = document.getElementById(`${metric}Chart`).getContext('2d'); if (charts[metric]) { charts[metric].destroy(); charts[metric] = null; } if (noData) return; const cfg = metaCfg[metric]; const allTimes = new Set([ ...sPoints.map(p => p.time), ...pPoints.map(p => p.time), ...rPoints.map(p => p.time) ]); const commonTimeline = Array.from(allTimes).sort(); const sensorMap = new Map(sPoints.map(p => [p.time, p.value])); const predMap = new Map(pPoints.map(p => [p.time, p.value])); const rawMap = new Map(rPoints.map(p => [p.time, p.value])); const sensorData = commonTimeline.map(t => ({ x: t, y: sensorMap.get(t) ?? null })); const predData = commonTimeline.map(t => ({ x: t, y: predMap.get(t) ?? null })); const rawData = commonTimeline.map(t => ({ x: t, y: rawMap.get(t) ?? null })); charts[metric] = new Chart(ctx, { type: 'line', data: { labels: commonTimeline, datasets: [ { label: 'Thực tế', data: sensorData, borderColor: cfg.color, backgroundColor: cfg.color + '18', borderWidth: isMain ? 2 : 1.5, fill: isMain, tension: 0.35, pointRadius: 0, spanGaps: true, }, { label: 'Dự đoán', data: predData, borderColor: '#ff4444', borderDash: [5, 4], borderWidth: isMain ? 1.5 : 1.2, tension: 0.35, pointRadius: 0, spanGaps: true, }, { label: 'Raw', data: rawData, borderColor: '#ffab40', borderDash: [2, 4], borderWidth: isMain ? 1.5 : 1.2, tension: 0.35, pointRadius: 0, spanGaps: true, } ] }, options: { animation: false, responsive: true, maintainAspectRatio: false, interaction: { mode: 'nearest', intersect: false, axis: 'x' }, plugins: { legend: { display: false } }, scales: { x: { type: 'category', grid: { color: '#1e2630' }, ticks: { color: '#506070', font: { family: 'JetBrains Mono', size: 9 }, maxRotation: 0, minRotation: 0, autoSkip: false, maxTicksLimit: 20, callback: function(value, index) { // hiển thị mỗi 4s if (index % 4 !== 0) return ''; const t = this.getLabelForValue(value); if (!t) return ''; const date = new Date(t); const h = String(date.getHours()).padStart(2, '0'); const m = String(date.getMinutes()).padStart(2, '0'); const s = String(date.getSeconds()).padStart(2, '0'); return `${h}h${m}p${s}s`; } } }, y: { min: cfg.yMin, max: cfg.yMax, grid: { color: '#1a2230', lineWidth: 0.5 }, ticks: { color: '#506070', font: { family: 'JetBrains Mono', size: 9 }, maxTicksLimit: 5 } } } } }); }
